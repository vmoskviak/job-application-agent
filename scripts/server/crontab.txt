# Crontab для сервера (root у LXC), TZ=Europe/Berlin.
# Встановлення: crontab /opt/jobsearch-autopilot/scripts/server/crontab.txt
#
# 20:30 — evening discover (дайджест, без подач) для всіх активних клієнтів
# 03:00 — morning apply (подачі + пошта + звіт) для всіх активних клієнтів
#
# Клієнти йдуть по черзі під одним локом; timeout одного рану 3h. З двома
# активними клієнтами ранкове вікно може тягнутись до ~09:00 — якщо це
# заважає денному ліміту підписки, зсунути старт раніше.
#
# Trial-режим (shadow): розкоментувати APPLY_MODE-рядок, закоментувати live.

SHELL=/bin/bash
PATH=/usr/local/bin:/usr/bin:/bin:/usr/local/sbin:/usr/sbin

30 20 * * * /opt/jobsearch-autopilot/scripts/server/run_all.sh evening
0 3 * * *   /opt/jobsearch-autopilot/scripts/server/run_all.sh morning
